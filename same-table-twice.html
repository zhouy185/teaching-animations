sql
SELECT
    pu.Borough AS pickup_borough,
    do.Borough AS dropoff_borough,
    COUNT(*) AS num_trips
FROM trips AS t
JOIN zones AS pu
    ON t.PULocationID = pu.LocationID
JOIN zones AS do
    ON t.DOLocationID = do.LocationID
GROUP BY
    pu.Borough,
    do.Borough
ORDER BY num_trips DESC
