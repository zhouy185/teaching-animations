sql
SELECT
    z.Borough AS pickup_borough,
    COUNT(*) AS num_trips,
    AVG(t.base_passenger_fare) AS avg_fare
FROM trips AS t
INNER JOIN zones AS z
    ON t.PULocationID = z.LocationID
GROUP BY z.Borough
ORDER BY num_trips DESC
